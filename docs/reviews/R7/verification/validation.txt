package unicode

public func width(char: Rune): Int8 {
    Codepoint(UInt32(char.position())).width
}

public func width(value: String): Int64 {
    var width = 0
    let ss = GraphemeIterator(value)  //cjlint-ignore !G.VAR.02
    while(true) {
        match(ss.next()) {
            case None => break
            case Some(g) => width += Int64(g.width)
        }
    }
    width
}

public interface UnicodeWidth {
    func width(): Int64
}

    public func width(): Int64 {
        Int64(width(this))
    }
}

    }
